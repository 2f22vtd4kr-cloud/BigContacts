   222	function candidateTypeToEntityType(type: string): "HNWI" | "Corporation" | "Trust" | "Gatekeeper" {
   223	  const normalized = type.toLowerCase();

   225	  if (normalized.includes("intermediary") || normalized.includes("gatekeeper") || normalized.includes("advisor")) return "Gatekeeper";

   230	function mergeContactEvidence(
   231	  ...groups: Array<Array<{
   232	    vectorType: string;
   233	    value: string;
   234	    scope: string;
   235	    personName: string | null;
   236	    role: string | null;
   237	    sourceUrls: string[];
   238	    note: string | null | undefined;
   239	  }> | undefined>
   240	) {
   241	  return groups.flatMap((group) => group ?? [])
   242	    .filter((item) => item.value.trim())
   243	    .filter((item, index, all) => all.findIndex((other) =>
   244	      other.vectorType === item.vectorType
   245	      && other.value.toLowerCase() === item.value.toLowerCase()
   246	      && (other.personName ?? "").toLowerCase() === (item.personName ?? "").toLowerCase(),
   247	    ) === index)
   248	    .slice(0, 20);
   249	}

   253	function mergeDiscoveryCandidates(
   254	  existing: DiscoveryCandidate[],
   255	  additions: DiscoveryCandidate[],
   256	): DiscoveryCandidate[] {
   257	  const merged = new Map<string, DiscoveryCandidate>();
   258	  for (const candidate of [...existing, ...additions]) {
   259	    const key = candidate.name.trim().toLowerCase();
   260	    if (!key) continue;
   261	    const prior = merged.get(key);
   262	    if (!prior) {
   263	      merged.set(key, candidate);
   264	      continue;
   265	    }
   266	    merged.set(key, {
   267	      ...prior,
   268	      type: prior.type || candidate.type,
   269	      relevance: candidate.relevance || prior.relevance,
   270	      reachability: candidate.reachability || prior.reachability,
   271	      sourceUrls: [...new Set([...prior.sourceUrls, ...candidate.sourceUrls])].slice(0, 20),
   272	      contactEvidence: mergeContactEvidence(prior.contactEvidence, candidate.contactEvidence),
   273	      admittedEntityId: prior.admittedEntityId ?? candidate.admittedEntityId ?? null,
   274	    });
   275	  }
   276	  // Fame-only trophies out; person-first ranking; shells kept only with reframe annotation.
   277	  return rankDiscoveryReviewCandidates(
   278	    filterDiscoveryCandidatesByFitness([...merged.values()]),
   279	  ).slice(0, 80);
   280	}

   405	      entityType === "HNWI"
   406	      || entityType === "Gatekeeper"

   442	          }
   443	        }
   444	      }
   445	    } catch {
   446	      // non-fatal
   447	    }
   448	  }
   449	
   450	  return { materialized, secondaryExpanded, candidates: updated };
   451	}
